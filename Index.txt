    const userIdNumerico = generarIdNumericoUnico();
    
    // 3. Actualizar displayName en Auth
    await updateProfile(user, { displayName: name });
    
    // 4. Preparar datos
    const userDataToSave = {
        uid: user.uid,
        userId: userIdNumerico,
        name: name,
        email: email,
        phone: phone,
        photoURL: null,
        saldo: 0,
        createdAt: new Date().toISOString(),
        role: 'user'
    };
    
    // 5. Procesar foto si existe
    if(photoInput.files[0]) {
        try {
            const photoBase64 = await compressImage(photoInput.files[0], 0.5, 400);
            userDataToSave.photoURL = photoBase64;
        } catch(e) {
            console.log('Error foto:', e);
        }
    }
    
    // 6. Guardar en Firestore (usando UID como ID de documento)
    await setDoc(doc(db, 'usuarios', user.uid), userDataToSave);
    
    alert('Cuenta creada exitosamente. Tu ID es: ' + userIdNumerico);
    
} catch(error) {
    console.error('Error registro:', error);
    let mensaje = 'Error: ';

        mensaje += 'Este correo ya está registrado';
    } else if(error.code === 'auth/weak-password') {
        mensaje += 'Contraseña débil (mín 6 caracteres)';
    } else {
        mensaje += error.message;
    }
    alert(mensaje);
} finally {
    submitBtn.disabled = false;
    submitBtn.innerHTML = '<i class="fas fa-user-plus"></i> CREAR CUENTA';
}

async function cargarDatosUsuario(){
    if(!currentUser) return;
    
    try {
        const userDocRef = doc(db, 'usuarios', currentUser.uid);
        const userDoc = await getDoc(userDocRef);
        
        if(userDoc.exists()) {
            userData = userDoc.data();
            
            // IMPORTANTE: Si el usuario antiguo no tiene userId, se lo creamos ahora
            if(!userData.userId) {
                const nuevoId = generarIdNumericoUnico();
                await updateDoc(userDocRef, { userId: nuevoId });
                userData.userId = nuevoId;
                console.log('ID asignado a usuario existente:', nuevoId);
            }
            
            // Sincronizar UID si es necesario
            if(userData.uid !== currentUser.uid) {
                await updateDoc(userDocRef, { uid: currentUser.uid });
                userData.uid = currentUser.uid;
            }
            
            actualizarUIPerfil();
        } else {
            // Crear documento si no existe (caso raro)
            const userIdNumerico = generarIdNumericoUnico();
            userData = {
                uid: currentUser.uid,
                userId: userIdNumerico,
                name: currentUser.displayName || 'Usuario',
                email: currentUser.email,
                phone: '',
                photoURL: null,
                saldo: 0,
                createdAt: new Date().toISOString(),
                role: 'user'
            };
            await setDoc(userDocRef, userData);
            actualizarUIPerfil();
        }
    } catch(error) {
        console.error('Error cargando datos:', error);
    }
}

async function cargarReviews(productId){

const snap=await getDocs(q);
const reviews=snap.docs.map(d=>d.data());

const total=reviews.length;
const avg=total>0?(reviews.reduce((a,b)=>a+b.rating,0)/total).toFixed(1):0;
document.getElementById('avgRating').textContent=avg;
document.getElementById('totalReviews').textContent=total+' reseñas';

const starsContainer=document.getElementById('avgStars');
starsContainer.innerHTML='';
for(let i=1;i<=5;i++){
if(i<=Math.round(avg)){
starsContainer.innerHTML+='<i class="fas fa-star"></i>';
}else if(i-0.5<=avg){
starsContainer.innerHTML+='<i class="fas fa-star-half-alt"></i>';
}else{
starsContainer.innerHTML+='<i class="far fa-star"></i>';
}
}

const counts=[0,0,0,0,0];
reviews.forEach(r=>counts[5-r.rating]++);
const max=Math.max(...counts)||1;
let barsHTML='';
for(let i=5;i>=1;i--){
const count=counts[5-i];
const percent=(count/max)*100;
barsHTML+=`
<div class="rating-bar">
<span>${i}★</span>
<div class="bar"><div class="fill" style="width:${percent}%"></div></div>
<span>${count}</span>
</div>
`;
}
document.getElementById('ratingBars').innerHTML=barsHTML;

const listContainer=document.getElementById('reviewsList');
if(total===0){
listContainer.innerHTML='<div style="text-align:center;padding:40px;color:var(--text-muted);">Sin reseñas aún. ¡Sé el primero!</div>';
return;
}

listContainer.innerHTML=reviews.map(r=>{

return `
<div class="review-card">

async function updateTopVip(){
try{
const comprasQuery=query(collection(db,"compras"),orderBy("timestamp","desc"));
const snap=await getDocs(comprasQuery);
const allCompras=snap.docs.map(d=>d.data());

const userTotals={};
for(const c of allCompras){
if(!userTotals[c.userId]){
userTotals[c.userId]={total:0,compras:0};
}
userTotals[c.userId].total+=c.total;
userTotals[c.userId].compras++;
}

const sorted=Object.entries(userTotals)
.map(([uid,data])=>({uid,...data}))
.sort((a,b)=>b.total-a.total)
.slice(0,5);

for(let i=0;i<sorted.length;i++){
const userDoc=await getDoc(doc(db,'usuarios',sorted[i].uid));
if(userDoc.exists()){
const uData=userDoc.data();
sorted[i].name=uData.name||'Usuario';
sorted[i].photo=uData.photoURL;
sorted[i].userId=uData.userId||'---';
}else{
sorted[i].name='Usuario';
sorted[i].userId='---';
}
}

const container=document.getElementById('topVipGrid');
const rankNumbers=[1,2,3,4,5];
const icons=['crown','medal','award','star','user'];

container.innerHTML=sorted.map((user,index)=>{
const rankClass=index===0?'rank-1':index===1?'rank-2':index===2?'rank-3':index===3?'rank-4':'rank-5';
const avatarContent=user.photo?
`<img src="${user.photo}">`:
`<i class="fas fa-${icons[index]}"></i>`;
const crown=index===0?'<i class="fas fa-crown vip-crown"></i>':'';

return `
<div class="vip-card ${rankClass}">
<div class="vip-rank">${rankNumbers[index]}</div>
<div class="vip-avatar">
${avatarContent}
${crown}
</div>
<div class="vip-name">${user.name}</div>
<div class="vip-amount">$${user.total.toFixed(2)}</div>
<div class="vip-label">TOP ${rankNumbers[index]}</div>
<div class="vip-stats">
<div class="vip-stat">
<div class="vip-stat-value">${user.compras}</div>
<div class="vip-stat-label">Compras</div>
</div>
<div class="vip-stat">

// IMAGE COMPRESSION
function compressImage(file,quality=0.45,maxWidth=800){
return new Promise((resolve,reject)=>{
const reader=new FileReader();
reader.onload=e=>{
const img=new Image();
img.onload=()=>{
const canvas=document.createElement("canvas");
let ratio=img.width/img.height;
let width=maxWidth;
let height=maxWidth/ratio;
if(img.width<maxWidth){
width=img.width;
height=img.height;
}
canvas.width=width;
canvas.height=height;
const ctx=canvas.getContext("2d");
ctx.drawImage(img,0,0,width,height);
resolve(canvas.toDataURL("image/webp",quality));
};
img.onerror=reject;
img.src=e.target.result;
};
reader.onerror=reject;
reader.readAsDataURL(file);
});
}

async function cargarTransferencias(){
if(!currentUser)return;
const q=query(collection(db,"transferencias"),where("fromUid","==",currentUser.uid));
const q2=query(collection(db,"transferencias"),where("toUid","==",currentUser.uid));
const [snap1,snap2]=await Promise.all([getDocs(q),getDocs(q2)]);
const transfers=[...snap1.docs.map(d=>({...d.data(),id:d.id,type:'sent'})),...snap2.docs.map(d=>({...d.data(),id:d.id,type:'received'}))];
transfers.sort((a,b)=>b.timestamp-a.timestamp);

const container=document.getElementById('transferHistoryList');
if(transfers.length===0){
container.innerHTML='<div style="text-align:center;padding:20px;color:var(--text-muted);">Sin transferencias aún</div>';
return;
}

container.innerHTML=transfers.map(t=>{
const fecha=new Date(t.timestamp).toLocaleDateString();
const isSent=t.type==='sent';
return `
<div class="transfer-item ${isSent?'sent':'received'}">
<div class="transfer-icon">
<i class="fas fa-${isSent?'paper-plane':'download'}"></i>
</div>
<div class="transfer-details">
<div class="type">${isSent?'Enviado':'Recibido'}</div>
<div class="user">${isSent?`Para: ${t.toName||t.toUserId}`:`De: ${t.fromName||t.fromUserId}`}</div>
<div class="date">${fecha}</div>
</div>
<div class="transfer-amount">${isSent?'-':'+'}$${parseFloat(t.amount).toFixed(2)}</div>
</div>
`;
}).join('');
}

window.realizarTransferencia=async function(){
const toInput=document.getElementById('transferTo').value.trim();
const amount=parseFloat(document.getElementById('transferAmount').value);
const message=document.getElementById('transferMessage').value.trim();

if(!toInput||!amount){
alert('Completa todos los campos');
return;
}

if(amount<1){
alert('Mínimo $1.00 USD');
return;
}

if(parseFloat(userData.saldo||0)<amount){
alert('Saldo insuficiente. Tienes $'+parseFloat(userData.saldo||0).toFixed(2)+' USD');
return;
}

try{
let destUid, destData, destUserId;

if(toInput.includes('@')){
// Buscar por email
const snap=await getDocs(query(collection(db,"usuarios"),where("email","==",toInput)));
if(snap.empty){
alert('Usuario no encontrado');
return;
}
destData=snap.docs[0].data();
destUid=snap.docs[0].id;
destUserId=destData.userId;
}else{
// Buscar por userId numérico
const snap=await getDocs(query(collection(db,"usuarios"),where("userId","==",toInput)));
if(snap.empty){
alert('Usuario no encontrado');
return;
}
destData=snap.docs[0].data();
destUid=snap.docs[0].id;
destUserId=destData.userId;
}

if(destUid===currentUser.uid){
alert('No puedes transferirte a ti mismo');
return;
}

// Realizar transferencia
const batch=writeBatch(db);

const nuevoSaldoOrigen = parseFloat((userData.saldo - amount).toFixed(2));
const nuevoSaldoDestino = parseFloat(((destData.saldo||0) + amount).toFixed(2));

// Actualizar saldos
batch.update(doc(db,"usuarios",currentUser.uid),{saldo:nuevoSaldoOrigen});
batch.update(doc(db,"usuarios",destUid),{saldo:nuevoSaldoDestino});

// Crear registro de transferencia
const transferRef=doc(collection(db,"transferencias"));
batch.set(transferRef,{
fromUid:currentUser.uid,
fromUserId:userData.userId,
fromName:userData.name,
toUid:destUid,
toUserId:destUserId,
toName:destData.name,
amount:parseFloat(amount.toFixed(2)),
message:message,
timestamp:Date.now()
});

await batch.commit();

alert(`Transferencia de $${amount.toFixed(2)} USD realizada exitosamente a ${destData.name}`);
document.getElementById('transferTo').value='';
document.getElementById('transferAmount').value='';
document.getElementById('transferMessage').value='';
cargarTransferencias();
}catch(e){
console.error(e);
alert('Error: '+e.message);
}
};
