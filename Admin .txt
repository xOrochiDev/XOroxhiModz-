onAuthStateChanged(auth, async (user) => {
    if (user) {
        currentUser = user;
        await cargarDatosAdmin(user);

async function cargarDatosAdmin(user) {
    try {
        const adminDoc = await getDoc(doc(db, 'admins', user.uid));
        const userDoc = await getDoc(doc(db, 'usuarios', user.uid));
        
        if (user.email === ADMIN_PRINCIPAL) {
            currentAdminData = {
                uid: user.uid,
                email: user.email,
                name: user.displayName || 'Admin Principal',
                role: SUPERADMIN_ROLE,
                permisos: {
                    productos: true,
                    agregarStock: true,
                    recargas: true,
                    saldo: true,
                    usuarios: true,
                    admins: true,
                    logs: true,
                    editarTodo: true
                },
                limiteDiario: 999999,
                limiteUsadoHoy: 0,
                createdAt: new Date().toISOString()
            };
            
            await setDoc(doc(db, 'admins', user.uid), currentAdminData, { merge: true });
            await setDoc(doc(db, 'usuarios', user.uid), {
                uid: user.uid,
                email: user.email,
                name: currentAdminData.name,
                role: ADMIN_ROLE
            }, { merge: true });
            
            mostrarPanel(user);
            return;
        }
        
        if (adminDoc.exists()) {
            currentAdminData = adminDoc.data();
            
            if (currentAdminData.estado === 'inactivo') {
                mostrarErrorAcceso('Tu cuenta ha sido desactivada. Contacta al admin principal.');
                await signOut(auth);
                return;
            }
            
            await verificarResetLimite();
            
            mostrarPanel(user);
        } else if (userDoc.exists() && userDoc.data().role === 'admin') {
            // Migrar de usuarios a admins
            currentAdminData = {
                uid: user.uid,
                email: user.email,
                name: userDoc.data().name || 'Administrador',
                role: ADMIN_ROLE,
                permisos: { agregarStock: true },
                limiteDiario: 100,
                limiteUsadoHoy: 0,
                createdAt: new Date().toISOString()
            };
            
            await setDoc(doc(db, 'admins', user.uid), currentAdminData);
            mostrarPanel(user);
        } else {
            mostrarErrorAcceso('No tienes permisos de administrador');
            await signOut(auth);
        }
    } catch (error) {
        console.error('Error cargando datos admin:', error);
        mostrarErrorAcceso('Error al verificar permisos');
    }
}

async function verificarResetLimite() {
    if (!currentAdminData) return;
    
    const hoy = new Date().toDateString();
    const ultimaActualizacion = currentAdminData.ultimaActualizacionLimite;
    
    if (ultimaActualizacion !== hoy) {
        await updateDoc(doc(db, 'admins', currentUser.uid), {
            limiteUsadoHoy: 0,
            ultimaActualizacionLimite: hoy
        });
        currentAdminData.limiteUsadoHoy = 0;
    }
}

function ocultarTabsSinPermiso() {
    const tabs = document.querySelectorAll('.tab[data-permiso]');

            tab.style.display = 'none';
        }
    });
}

function tienePermiso(permiso) {
    if (!currentAdminData) return false;
    if (currentAdminData.role === SUPERADMIN_ROLE) return true;

}

window.doLogin = async function() {
    const email = document.getElementById('loginEmail').value;
    const pass = document.getElementById('loginPass').value;
    const btn = document.getElementById('btnLogin');
    
    if (!email || !pass) {
        mostrarErrorAcceso('Ingresa correo y contraseña');
        return;
    }
    
    btn.disabled = true;
    btn.innerHTML = '<i class="fas fa-circle-notch fa-spin"></i> Verificando...';
    
    try {
        await signInWithEmailAndPassword(auth, email, pass);
    } catch (error) {
        console.error(error);
        let msg = 'Error al iniciar sesión';
        if (error.code === 'auth/user-not-found') msg = 'Usuario no encontrado';
        if (error.code === 'auth/wrong-password') msg = 'Contraseña incorrecta';
        if (error.code === 'auth/invalid-email') msg = 'Correo inválido';
        if (error.code === 'auth/invalid-credential') msg = 'Credenciales inválidas';
        mostrarErrorAcceso(msg);
    } finally {
        btn.disabled = false;
        btn.innerHTML = '<i class="fas fa-sign-in-alt"></i> Ingresar';
    }
};

window.aprobarRecarga = async function(firebaseId, index, userUid) {
    if (!tienePermiso('recargas')) {
        alert('No tienes permiso para aprobar recargas');
        return;
    }
    
    try {
        const monto = parseFloat(document.getElementById(`saldoCustom${index}`).value);
        
        if (!monto || monto <= 0) {
            alert('Ingresa un monto válido');
            return;
        }

        if (currentAdminData.role !== SUPERADMIN_ROLE) {
            const limiteRestante = currentAdminData.limiteDiario - (currentAdminData.limiteUsadoHoy || 0);
            if (monto > limiteRestante) {
                alert(`Has excedido tu límite diario. Restante: $${limiteRestante.toFixed(2)}`);
                return;
            }
        }

        let uidReal = userUid;
        if (!usuarios[userUid] || !usuarios[userUid].uid) {
            const snap = await getDocs(query(collection(db, 'usuarios'), where('userId', '==', userUid)));
            if (!snap.empty) {
                uidReal = snap.docs[0].id;
            }
        }

        const ref = doc(db, 'usuarios', uidReal);
        const snap = await getDoc(ref);

        if (!snap.exists()) {
            alert('Usuario no existe');
            return;
        }

        const data = snap.data();
        const saldoActual = parseFloat(data.saldo || 0);
        const nuevoSaldo = parseFloat((saldoActual + monto).toFixed(2));

        await updateDoc(ref, { saldo: nuevoSaldo });
        
        await updateDoc(doc(db, 'recargas', firebaseId), { 
            estado: 'aprobada',
            montoAprobado: monto,
            aprobadoPor: currentUser.uid,
            aprobadoPorName: currentAdminData.name || currentUser.email,
            fechaAprobacion: new Date().toISOString()
        });

        if (currentAdminData.role !== SUPERADMIN_ROLE) {
            const nuevoUsado = (currentAdminData.limiteUsadoHoy || 0) + monto;
            await updateDoc(doc(db, 'admins', currentUser.uid), {
                limiteUsadoHoy: nuevoUsado
            });
            currentAdminData.limiteUsadoHoy = nuevoUsado;
            actualizarIndicadorLimite();
        }

        await registrarLog('RECARGA_APROBADA', {
            recargaId: firebaseId,
            usuario: uidReal,
            monto,
            aprobadoPor: currentUser.email
        }, 'success');

        alert(`Saldo de $${monto.toFixed(2)} acreditado correctamente`);
    } catch (e) {
        console.error(e);
        alert('Error al aprobar: ' + e.message);
    }
};

function actualizarIndicadorLimite() {
    if (!currentAdminData || currentAdminData.role === SUPERADMIN_ROLE) return;
    
    const usado = currentAdminData.limiteUsadoHoy || 0;
    const total = currentAdminData.limiteDiario || 100;
    const restante = total - usado;
    const porcentaje = (usado / total) * 100;
    
    document.getElementById('statLimite').textContent = `$${restante.toFixed(2)}`;
    document.getElementById('limiteUsado').textContent = `$${usado.toFixed(2)}`;
    document.getElementById('limiteTotal').textContent = `$${total.toFixed(2)}`;
    
    const progress = document.getElementById('limiteProgress');
    progress.style.width = `${Math.min(porcentaje, 100)}%`;
    
    if (porcentaje > 90) {
        progress.className = 'limite-progress danger';
    } else if (porcentaje > 70) {
        progress.className = 'limite-progress warning';
    } else {
        progress.className = 'limite-progress';
    }
}

async function cambiarSaldo(tipo) {
    if (!tienePermiso('saldo')) {
        alert('No tienes permiso para gestionar saldo');
        return;
    }
    
    try {
        const idInput = document.getElementById('idUsuarioSaldo').value.trim();
        const monto = parseFloat(document.getElementById('montoSaldo').value);
        const motivo = document.getElementById('motivoSaldo').value.trim();

        if (!idInput || !monto || monto <= 0) {
            alert('Completa todos los campos con valores válidos');
            return;
        }

        if (currentAdminData.role !== SUPERADMIN_ROLE) {
            const limiteRestante = currentAdminData.limiteDiario - (currentAdminData.limiteUsadoHoy || 0);
            if (monto > limiteRestante) {
                alert(`Has excedido tu límite diario. Restante: $${limiteRestante.toFixed(2)}`);
                return;
            }
        }

        let uid = null;
        let userData = null;

        if (idInput.includes('@')) {
            const snap = await getDocs(query(collection(db, 'usuarios'), where('email', '==', idInput)));
            if (!snap.empty) {
                uid = snap.docs[0].id;
                userData = snap.docs[0].data();
            }
        } else {
            const snap = await getDocs(query(collection(db, 'usuarios'), where('userId', '==', idInput)));
            if (!snap.empty) {
                uid = snap.docs[0].id;
                userData = snap.docs[0].data();
            } else {
                const docSnap = await getDoc(doc(db, 'usuarios', idInput));
                if (docSnap.exists()) {
                    uid = idInput;
                    userData = docSnap.data();
                }
            }
        }

        if (!uid) {
            alert('Usuario no encontrado');
            return;
        }

        const ref = doc(db, 'usuarios', uid);
        const saldoActual = parseFloat(userData.saldo || 0);
        let nuevoSaldo;

        if (tipo === 'quitar') {
            if (saldoActual < monto) {
                alert('El usuario no tiene saldo suficiente');
                return;
            }
            nuevoSaldo = parseFloat((saldoActual - monto).toFixed(2));
        } else {
            nuevoSaldo = parseFloat((saldoActual + monto).toFixed(2));
        }

        await updateDoc(ref, { saldo: nuevoSaldo });
        
        if (currentAdminData.role !== SUPERADMIN_ROLE && tipo === 'agregar') {
            const nuevoUsado = (currentAdminData.limiteUsadoHoy || 0) + monto;
            await updateDoc(doc(db, 'admins', currentUser.uid), {
                limiteUsadoHoy: nuevoUsado
            });
            currentAdminData.limiteUsadoHoy = nuevoUsado;
            actualizarIndicadorLimite();
        }
        
        await registrarLog(tipo === 'agregar' ? 'SALDO_AGREGADO' : 'SALDO_DESCONTADO', {
            usuario: uid,
            usuarioEmail: userData.email,
            monto,
            motivo: motivo || 'Sin motivo'
        }, tipo === 'agregar' ? 'success' : 'warning');
        
        document.getElementById('idUsuarioSaldo').value = '';
        document.getElementById('montoSaldo').value = '';
        document.getElementById('motivoSaldo').value = '';
        
        alert(`Saldo ${tipo === 'agregar' ? 'agregado' : 'descontado'} correctamente. Nuevo saldo: $${nuevoSaldo.toFixed(2)}`);
    } catch (e) {
        console.error(e);
        alert('Error al actualizar saldo: ' + e.message);
    }
}

window.togglePermiso = function(element, permiso) {
    const icon = element.querySelector('.fa-check-circle');
    const isActive = element.classList.contains('activo');
    
    if (isActive) {
        element.classList.remove('activo');
        icon.style.opacity = '0';
        permisosNuevoAdmin[permiso] = false;
    } else {
        element.classList.add('activo');
        icon.style.opacity = '1';
        permisosNuevoAdmin[permiso] = true;
    }
};

window.crearNuevoAdmin = async function() {
    if (currentAdminData.role !== SUPERADMIN_ROLE) {
        alert('Solo el admin principal puede crear admins');
        return;
    }
    
    const email = document.getElementById('nuevoAdminEmail').value;
    const pass = document.getElementById('nuevoAdminPass').value;
    const nombre = document.getElementById('nuevoAdminNombre').value;
    const limite = parseFloat(document.getElementById('nuevoAdminLimite').value) || 100;
    
    if (!email || !pass || !nombre) {
        alert('Completa todos los campos');
        return;
    }
    
    if (pass.length < 6) {
        alert('La contraseña debe tener al menos 6 caracteres');
        return;
    }
    
    // CORRECCIÓN DEL BUG: Solo usar los permisos seleccionados, sin defaults
    // Eliminar todos los permisos que no fueron explícitamente seleccionados
    const permisosFinales = {};
    
    // Solo incluir permisos que fueron activados explícitamente
    Object.keys(permisosNuevoAdmin).forEach(key => {
        if (permisosNuevoAdmin[key] === true) {
            permisosFinales[key] = true;
        }
    });
    
    // Si no se seleccionó ningún permiso, mostrar error
    if (Object.keys(permisosFinales).length === 0) {
        alert('Debes seleccionar al menos un permiso');
        return;
    }
    
    try {
        const userCredential = await createUserWithEmailAndPassword(auth, email, pass);
        const user = userCredential.user;
        
        await setDoc(doc(db, 'usuarios', user.uid), {
            uid: user.uid,
            userId: Date.now().toString().slice(-6) + Math.floor(10 + Math.random() * 90),
            email: email,
            name: nombre,
            phone: '',
            photoURL: null,
            saldo: 0,
            role: 'admin',
            createdAt: new Date().toISOString()
        });
        
        await setDoc(doc(db, 'admins', user.uid), {
            uid: user.uid,
            email: email,
            name: nombre,
            role: ADMIN_ROLE,
            permisos: permisosFinales,  // Solo los permisos seleccionados explícitamente
            limiteDiario: limite,
            limiteUsadoHoy: 0,
            ultimaActualizacionLimite: new Date().toDateString(),
            estado: 'activo',
            createdAt: new Date().toISOString(),
            createdBy: currentUser.uid
        });
        
        await updateProfile(user, { displayName: nombre });
        
        await registrarLog('ADMIN_CREADO', {
            nuevoAdmin: email,
            permisos: permisosFinales,
            limiteDiario: limite
        }, 'success');
        
        alert('Nuevo admin creado correctamente con permisos: ' + Object.keys(permisosFinales).join(', '));
        
        document.getElementById('nuevoAdminEmail').value = '';
        document.getElementById('nuevoAdminPass').value = '';
        document.getElementById('nuevoAdminNombre').value = '';
        document.getElementById('nuevoAdminLimite').value = '100';
        document.querySelectorAll('.permiso-item').forEach(el => {
            el.classList.remove('activo');
            el.querySelector('.fa-check-circle').style.opacity = '0';
        });
        permisosNuevoAdmin = {};
    } catch (error) {
        console.error(error);
        alert('Error: ' + error.message);
    }
};

window.editarAdmin = async function(adminId) {
    if (currentAdminData.role !== SUPERADMIN_ROLE) {
        alert('Solo el admin principal puede editar admins');
        return;
    }
    
    const admin = admins.find(a => a.uid === adminId);
    if (!admin) return;
    
    const nuevoLimite = prompt('Nuevo límite diario (actual: $' + (admin.limiteDiario || 100) + '):', admin.limiteDiario || 100);
    if (nuevoLimite === null) return;
    
    try {
        await updateDoc(doc(db, 'admins', adminId), {
            limiteDiario: parseFloat(nuevoLimite) || 100
        });
        
        await registrarLog('ADMIN_EDITADO', {
            admin: admin.email,
            nuevoLimite: parseFloat(nuevoLimite)
        }, 'warning');
        
        alert('Límite actualizado');
    } catch (e) {
        alert('Error: ' + e.message);
    }
};

// NUEVA FUNCIÓN: Guardar stock desde vista solo stock
window.guardarStockSolo = async function(productoId, index) {
    try {
        const producto = productos.find(p => p.firebaseId === productoId);
        if (!producto) return;
        
        const nuevasKeys = [];
        
        producto.precios.forEach((p, x) => {
            const val = document.getElementById(`stockKeys${index}_${x}`).value;
            const keysArray = val.split(',').map(y => y.trim()).filter(Boolean);
            nuevasKeys.push({
                dias: p.dias,
                precio: p.precio,
                keys: keysArray
            });
        });

        await updateDoc(doc(db, 'productos', productoId), { 
            precios: nuevasKeys,
            updatedAt: Date.now(),
            updatedBy: currentUser.uid
        });
        
        await registrarLog('STOCK_AGREGADO', {
            productoId: productoId,
            nombre: producto.nombre
        }, 'success');
        
        alert('Stock agregado correctamente');
    } catch (e) {
        console.error(e);
        alert('Error al agregar stock: ' + e.message);
    }
};

                        <div style="font-size: 13px; color: var(--text-muted);">${a.email}</div>
                        <div style="margin-top: 4px;">
                            ${isSuperAdmin ? '<span class="rango-badge rango-superadmin" style="font-size: 10px;"><i class="fas fa-crown"></i> PRINCIPAL</span>' : 
                              a.estado === 'inactivo' ? '<span class="tag tag-danger" style="font-size: 10px;">INACTIVO</span>' : 
                              '<span class="rango-badge rango-admin" style="font-size: 10px;">ADMIN</span>'}
                            ${permisosHTML}
                        </div>
                        ${!isSuperAdmin ? `<div style="font-size: 12px; color: var(--text-muted); margin-top: 4px;">Límite: $${a.limiteDiario || 100}/día • Usado hoy: $${a.limiteUsadoHoy || 0}</div>` : ''}
                    </div>
                    ${puedeEditar ? `
                    <div style="display: flex; gap: 8px;">
                        <button class="btn btn-warning btn-sm" onclick="editarAdmin('${a.uid}')" title="Editar límite">
                            <i class="fas fa-edit"></i>
                        </button>
                        <button class="btn ${a.estado === 'inactivo' ? 'btn-success' : 'btn-danger'} btn-sm" onclick="cambiarEstadoAdmin('${a.uid}', '${a.estado === 'inactivo' ? 'activo' : 'inactivo'}')" title="${a.estado === 'inactivo' ? 'Activar' : 'Desactivar'}">
                            <i class="fas fa-${a.estado === 'inactivo' ? 'check' : 'ban'}"></i>
                        </button>
                    </div>
                    ` : ''}
                </div>
            </div>
        `;
    });

    container.innerHTML = html;
}

function renderLogs() {
    const container = document.getElementById('listaLogs');
    
    if (!logs.length) {
        container.innerHTML = `
            <div class="empty-state">
                <i class="fas fa-history"></i>
                <p>No hay registros de actividad</p>
            </div>
        `;
        return;
    }

    let html = '';
    logs.slice(0, 100).forEach(log => {
        const iconClass = log.tipo || 'info';
        const iconMap = {
            success: 'check-circle',
            warning: 'exclamation-triangle',
            danger: 'times-circle',
            info: 'info-circle'
        };
        
        const detalles = log.detalles ? JSON.parse(log.detalles) : {};
        
        html += `
            <div class="log-entry">
                <div class="log-icon ${iconClass}">
                    <i class="fas fa-${iconMap[iconClass]}"></i>
                </div>
                <div class="log-content">
                    <div class="log-title">${log.accion}</div>
                    <div class="log-meta">
                        Por: ${log.adminName || log.adminEmail || 'Desconocido'} • 
                        ${new Date(log.timestamp).toLocaleString()}
                    </div>
                    ${Object.keys(detalles).length ? `
                    <div style="margin-top: 8px; padding: 8px; background: var(--bg-tertiary); border-radius: 8px; font-size: 12px; color: var(--text-secondary);">
                        ${Object.entries(detalles).map(([k, v]) => `${k}: ${v}`).join(' • ')}
                    </div>
                    ` : ''}
                </div>
            </div>
        `;
    });

    container.innerHTML = html;
}

function renderUsuariosRecientes() {
    const container = document.getElementById('listaUsuariosRecientes');
    
    const usuariosList = Object.entries(usuarios)
        .filter(([id, u]) => u.userId)
        .sort((a, b) => (b[1].createdAt || 0) - (a[1].createdAt || 0))
        .slice(0, 10);
    
    if (!usuariosList.length) {
        container.innerHTML = `
            <div class="empty-state">
                <i class="fas fa-users"></i>
                <p>No hay usuarios registrados</p>
            </div>
        `;
        return;
    }

    let html = '';
    usuariosList.forEach(([id, u]) => {
        html += `
            <div class="card animate-in" style="padding: 16px;">
                <div style="display: flex; align-items: center; gap: 12px;">
                    <div style="width: 40px; height: 40px; background: var(--bg-tertiary); border-radius: 50%; display: flex; align-items: center; justify-content: center; color: var(--accent);">
                        <i class="fas fa-user"></i>
                    </div>
                    <div style="flex: 1;">
                        <div style="font-weight: 600; color: var(--text-primary);">${u.name || 'Sin nombre'}</div>
                        <div style="font-size: 12px; color: var(--text-muted);">${u.email}</div>
                    </div>
                    <div style="text-align: right;">
                        <div style="font-size: 12px; color: var(--accent); font-weight: 600;">ID: ${u.userId}</div>
                        <div style="font-size: 11px; color: var(--text-muted);">$${parseFloat(u.saldo || 0).toFixed(2)}</div>
                    </div>
                </div>
            </div>
        `;
    });

    container.innerHTML = html;
}

function actualizarStats() {
    document.getElementById('statProductos').innerText = productos.length;
    
    const pendientes = recargas.filter(r => r.estado === 'pendiente');
    document.getElementById('statRecargas').innerText = pendientes.length;
    document.getElementById('badgeRecargas').innerText = pendientes.length;
    document.getElementById('badgeRecargas').style.display = pendientes.length > 0 ? 'inline-flex' : 'none';
    
    document.getElementById('statCompras').innerText = compras.length;
    
    const totalUsuarios = Object.keys(usuarios).filter(k => usuarios[k].userId).length;
    document.getElementById('statUsuarios').innerText = totalUsuarios;
}

function verificarStockBajo() {
    const alertas = [];
    productos.forEach(p => {
        p.precios.forEach((pr, idx) => {
            const stock = pr.keys ? pr.keys.length : 0;
            if (stock < 3 && stock > 0) {
                alertas.push({
                    producto: p.nombre,
                    dias: pr.dias,
                    stock: stock
                });
            } else if (stock === 0) {
                alertas.push({
                    producto: p.nombre,
                    dias: pr.dias,
                    stock: 0,
                    agotado: true
                });
            }
        });
    });
    
    const container = document.getElementById('alertasStock');
    if (alertas.length === 0) {
        container.innerHTML = `
            <div class="empty-state">
                <i class="fas fa-check-circle" style="color: var(--accent);"></i>
                <p>Todo en orden</p>
            </div>
        `;
    } else {
        let html = '';
        alertas.forEach(a => {
            const icon = a.agotado ? 'times' : 'exclamation-triangle';
            const color = a.agotado ? 'var(--danger)' : 'var(--warning)';
            const texto = a.agotado ? 'AGOTADO' : `${a.stock} llaves`;
            
            html += `
                <div class="card" style="border-left: 3px solid ${color}; animation: slideIn 0.3s ease;">
                    <div style="display: flex; align-items: center; gap: 12px;">
                        <i class="fas fa-${icon}" style="color: ${color}; font-size: 20px;"></i>
                        <div>
                            <div style="font-weight: 600; color: var(--text-primary);">${a.producto}</div>
                            <div style="font-size: 13px; color: var(--text-muted);">
                                ${a.dias} días - <span style="color: ${color}; font-weight: 600;">${texto}</span>
                            </div>
                        </div>
                    </div>
                </div>
            `;
        });
        container.innerHTML = html;
    }
}

// ============ INICIALIZACIÓN ============

function iniciarApp() {
    // Escuchar productos
    onSnapshot(collection(db, 'productos'), snap => {
        productos = snap.docs.map(d => ({ firebaseId: d.id, ...d.data() }));
        renderProductos();
        renderStockSolo(); // También actualizar vista de solo stock
        actualizarStats();
        verificarStockBajo();
    });

    // Escuchar compras
    onSnapshot(query(collection(db, 'compras'), orderBy('timestamp', 'desc')), snap => {
        compras = snap.docs.map(d => ({ firebaseId: d.id, ...d.data() }));
        renderCompras();
        actualizarStats();
    });

    // Escuchar recargas
    onSnapshot(collection(db, 'recargas'), snap => {
        recargas = snap.docs.map(d => ({ firebaseId: d.id, ...d.data() }));
        renderRecargas();
        actualizarStats();
    });

    // Escuchar usuarios
    onSnapshot(collection(db, 'usuarios'), snap => {
        usuarios = {};
        snap.forEach(d => { 
            usuarios[d.id] = d.data();
            if (d.data().userId) {
                usuarios[d.data().userId] = d.data();
            }
        });
        actualizarStats();
        renderUsuariosRecientes();
    });

    // Escuchar admins
    onSnapshot(collection(db, 'admins'), snap => {
        admins = snap.docs.map(d => ({ uid: d.id, ...d.data() }));
        renderAdmins();
    });

    // Escuchar logs (solo si tiene permiso)
    if (tienePermiso('logs')) {
        onSnapshot(query(collection(db, 'logs'), orderBy('timestamp', 'desc'), limit(100)), snap => {
            logs = snap.docs.map(d => d.data());
            renderLogs();
        });
    } else {
        document.getElementById('listaLogs').innerHTML = `
            <div class="access-denied">
                <i class="fas fa-lock"></i>
                <h3>Acceso Denegado</h3>
                <p>No tienes permiso para ver los logs</p>
            </div>
        `;
    }
}
